cmake_minimum_required(VERSION 3.16)
project(TradingResearchPortable LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(research_native SHARED src/research_native.cpp)
set_target_properties(research_native PROPERTIES CXX_VISIBILITY_PRESET hidden)
add_executable(native_test tests/native_test.cpp)
target_link_libraries(native_test PRIVATE research_native)
enable_testing()
add_test(NAME native_causal_and_validation COMMAND native_test)
